/**
 * Hook for Google Docs Integration
 * 
 * Setup:
 * 1. Add Google Docs API scope to GOOGLE_CONFIG
 * 2. Configure OAuth credentials
 * 3. Call googleAuth to get access token
 * 4. Use this hook to create/edit documents
 */

import { useState } from 'react';
import { GOOGLE_CONFIG } from '../config/google';

interface GoogleDocResponse {
  docId: string;
  docUrl: string;
  title: string;
}

export const useGoogleDocs = () => {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  /**
   * Create a new Google Document
   * @param title - Title of the document
   * @param accessToken - Google OAuth access token
   * @returns Document ID and URL
   */
  const createDocument = async (
    title: string,
    accessToken: string
  ): Promise<GoogleDocResponse | null> => {
    setLoading(true);
    setError(null);

    try {
      // Call Lambda function to create Google Doc
      // Lambda will use: POST ${GOOGLE_CONFIG.endpoints.DOCS}/documents
      // With headers: Authorization: Bearer {accessToken}
      // Body: { title }
      
      const response = await fetch('/api/google/docs/create', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${accessToken}`
        },
        body: JSON.stringify({ 
          title,
          // PLACEHOLDER: Google OAuth Access Token will be used by Lambda
          // Lambda endpoint: POST /api/google/docs/create
        })
      });

      if (!response.ok) throw new Error('Failed to create document');
      
      const data = await response.json();
      return {
        docId: data.documentId,
        docUrl: `https://docs.google.com/document/d/${data.documentId}/edit`,
        title: data.title
      };
    } catch (err) {
      const errorMessage = err instanceof Error ? err.message : 'Unknown error';
      setError(errorMessage);
      return null;
    } finally {
      setLoading(false);
    }
  };

  /**
   * Add content to an existing Google Document
   * @param docId - Google Document ID
   * @param content - Content to add
   * @param accessToken - Google OAuth access token
   */
  const appendContent = async (
    docId: string,
    content: string,
    accessToken: string
  ): Promise<boolean> => {
    setLoading(true);
    setError(null);

    try {
      // Call Lambda function to append to Google Doc
      // Lambda will use: PATCH ${GOOGLE_CONFIG.endpoints.DOCS}/documents/{docId}:batchUpdate
      // With headers: Authorization: Bearer {accessToken}
      // Body: { requests: [{ insertText: { text: content } }] }
      
      const response = await fetch(`/api/google/docs/${docId}/append`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${accessToken}`
        },
        body: JSON.stringify({ content })
      });

      return response.ok;
    } catch (err) {
      const errorMessage = err instanceof Error ? err.message : 'Unknown error';
      setError(errorMessage);
      return false;
    } finally {
      setLoading(false);
    }
  };

  /**
   * Share a Google Document with users
   * @param docId - Google Document ID
   * @param emails - Array of email addresses
   * @param accessToken - Google OAuth access token
   * @param permission - 'reader' | 'writer' | 'commenter'
   */
  const shareDocument = async (
    docId: string,
    emails: string[],
    accessToken: string,
    permission: 'reader' | 'writer' | 'commenter' = 'reader'
  ): Promise<boolean> => {
    setLoading(true);
    setError(null);

    try {
      // Call Lambda function to share Google Doc
      // Lambda will use: POST ${GOOGLE_CONFIG.endpoints.DRIVE}/files/{docId}/permissions
      // With headers: Authorization: Bearer {accessToken}
      // Body: { emailAddress, role, type: 'user' }
      
      const response = await fetch(`/api/google/docs/${docId}/share`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${accessToken}`
        },
        body: JSON.stringify({ emails, permission })
      });

      return response.ok;
    } catch (err) {
      const errorMessage = err instanceof Error ? err.message : 'Unknown error';
      setError(errorMessage);
      return false;
    } finally {
      setLoading(false);
    }
  };

  return { createDocument, appendContent, shareDocument, loading, error };
};

// src/screens/Calendar.tsx
import React, { useState, useEffect } from 'react';
import { useGoogleCalendar } from '../hooks/useGoogleCalendar';
import { useAuth } from '../hooks/useAuth';
import { CalendarEvent } from '../types';

const Calendar: React.FC = () => {
  const { user } = useAuth();
  const { createEvent, listEvents, loading, error } = useGoogleCalendar();
  const [events, setEvents] = useState<CalendarEvent[]>([]);
  const [showForm, setShowForm] = useState(false);
  const [formData, setFormData] = useState({
    title: '',
    startDate: '',
    endDate: '',
    location: '',
    description: ''
  });

  useEffect(() => {
    fetchEvents();
  }, [user]);

  const fetchEvents = async () => {
    try {
      // Call Lambda function to fetch calendar events from DynamoDB
      // Lambda endpoint: GET /api/calendar/events
      const response = await fetch('/api/calendar/events', {
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('accessToken')}`
        }
      });

      if (response.ok) {
        const data = await response.json();
        setEvents(data);
      }
    } catch (err) {
      console.error('Failed to fetch events:', err);
    }
  };

  const handleCreateEvent = async () => {
    if (!formData.title.trim() || !formData.startDate) {
      alert('Please fill in required fields');
      return;
    }

    try {
      const googleAccessToken = localStorage.getItem('googleAccessToken');
      if (!googleAccessToken) {
        alert('Please authenticate with Google first');
        return;
      }

      // Create event in Google Calendar
      const eventResult = await createEvent(
        {
          id: '',
          title: formData.title,
          startDate: formData.startDate,
          endDate: formData.endDate || formData.startDate,
          location: formData.location,
          description: formData.description
        },
        googleAccessToken
      );

      if (!eventResult) throw new Error('Failed to create event');

      // Save event to DynamoDB
      const response = await fetch('/api/calendar/events', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${localStorage.getItem('accessToken')}`
        },
        body: JSON.stringify({
          title: formData.title,
          startDate: formData.startDate,
          endDate: formData.endDate || formData.startDate,
          location: formData.location,
          description: formData.description,
          googleCalendarId: eventResult.eventId,
          createdBy: user?.email
        })
      });

      if (response.ok) {
        const newEvent = await response.json();
        setEvents([...events, newEvent]);
        setFormData({
          title: '',
          startDate: '',
          endDate: '',
          location: '',
          description: ''
        });
        setShowForm(false);
      }
    } catch (err) {
      alert(`Error: ${err instanceof Error ? err.message : 'Unknown error'}`);
    }
  };

  if (!user) {
    return <div className="p-6 text-center text-gray-600">Please log in</div>;
  }

  return (
    <div className="p-6 max-w-4xl mx-auto">
      <div className="flex justify-between items-center mb-6">
        <h1 className="text-3xl font-bold">Calendar</h1>
        <button
          onClick={() => setShowForm(!showForm)}
          className="bg-blue-600 text-white px-4 py-2 rounded hover:bg-blue-700"
        >
          + New Event
        </button>
      </div>

      {error && (
        <div className="mb-6 p-4 bg-red-100 border border-red-400 text-red-700 rounded">
          {error}
        </div>
      )}

      {showForm && (
        <div className="mb-6 p-4 border rounded bg-gray-50">
          <div className="mb-4">
            <label className="block text-sm font-medium mb-2">Event Title *</label>
            <input
              type="text"
              value={formData.title}
              onChange={(e) => setFormData({ ...formData, title: e.target.value })}
              placeholder="Event title"
              className="w-full px-3 py-2 border rounded"
            />
          </div>

          <div className="grid grid-cols-2 gap-4 mb-4">
            <div>
              <label className="block text-sm font-medium mb-2">Start Date *</label>
              <input
                type="datetime-local"
                value={formData.startDate}
                onChange={(e) => setFormData({ ...formData, startDate: e.target.value })}
                className="w-full px-3 py-2 border rounded"
              />
            </div>
            <div>
              <label className="block text-sm font-medium mb-2">End Date</label>
              <input
                type="datetime-local"
                value={formData.endDate}
                onChange={(e) => setFormData({ ...formData, endDate: e.target.value })}
                className="w-full px-3 py-2 border rounded"
              />
            </div>
          </div>

          <div className="mb-4">
            <label className="block text-sm font-medium mb-2">Location</label>
            <input
              type="text"
              value={formData.location}
              onChange={(e) => setFormData({ ...formData, location: e.target.value })}
              placeholder="Event location"
              className="w-full px-3 py-2 border rounded"
            />
          </div>

          <div className="mb-4">
            <label className="block text-sm font-medium mb-2">Description</label>
            <textarea
              value={formData.description}
              onChange={(e) => setFormData({ ...formData, description: e.target.value })}
              placeholder="Event description"
              rows={4}
              className="w-full px-3 py-2 border rounded"
            />
          </div>

          <div className="flex gap-2">
            <button
              onClick={handleCreateEvent}
              disabled={loading}
              className="bg-blue-600 text-white px-4 py-2 rounded hover:bg-blue-700 disabled:opacity-50"
            >
              {loading ? 'Creating...' : 'Create Event'}
            </button>
            <button
              onClick={() => setShowForm(false)}
              className="bg-gray-400 text-white px-4 py-2 rounded hover:bg-gray-500"
            >
              Cancel
            </button>
          </div>
        </div>
      )}

      {events.length === 0 ? (
        <div className="text-center p-8 border rounded bg-gray-50">
          <p className="text-gray-600">No upcoming events</p>
        </div>
      ) : (
        <div className="grid gap-4">
          {events.map((event) => (
            <div key={event.id} className="p-4 border rounded hover:shadow-lg transition-shadow">
              <h3 className="font-semibold text-lg">{event.title}</h3>
              <p className="text-sm text-gray-600">
                {new Date(event.startDate).toLocaleDateString()} {new Date(event.startDate).toLocaleTimeString()}
              </p>
              {event.location && <p className="text-sm text-gray-600">📍 {event.location}</p>}
              {event.description && <p className="text-gray-700 mt-2">{event.description}</p>}
            </div>
          ))}
        </div>
      )}
    </div>
  );
};
