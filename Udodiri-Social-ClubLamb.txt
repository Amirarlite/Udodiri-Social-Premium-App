/**
 * AWS Lambda - Authentication Handler
 * Handles user login, registration, logout, and token verification via AWS Cognito
 * 
 * Environment Variables Required:
 * - COGNITO_USER_POOL_ID
 * - COGNITO_CLIENT_ID
 * - REGION
 */

import {
  CognitoIdentityServiceProvider,
  DynamoDB
} from 'aws-sdk';
import crypto from 'crypto';

const cognito = new CognitoIdentityServiceProvider();
const dynamodb = new DynamoDB.DocumentClient();

interface AuthRequest {
  email: string;
  password: string;
  name?: string;
}

/**
 * Register new user in Cognito User Pool
 */
export const register = async (event: any) => {
  try {
    const { email, password, name } = JSON.parse(event.body) as AuthRequest;

    if (!email || !password) {
      return {
        statusCode: 400,
        body: JSON.stringify({ error: 'Email and password required' })
      };
    }

    // Sign up user in Cognito
    const params = {
      ClientId: process.env.COGNITO_CLIENT_ID!,
      Username: email,
      Password: password,
      UserAttributes: [
        { Name: 'email', Value: email },
        { Name: 'name', Value: name || email }
      ]
    };

    await cognito.signUp(params).promise();

    // Create user record in DynamoDB
    await dynamodb.put({
      TableName: process.env.USERS_TABLE!,
      Item: {
        id: crypto.randomUUID(),
        email,
        name: name || email,
        subscriptionTier: 'free',
        role: 'Member',
        level: 'Level 1',
        photo: 'https://i.postimg.cc/bJQgWxd8/udodiri-young-social-club.jpg',
        createdAt: new Date().toISOString()
      }
    }).promise();

    return {
      statusCode: 201,
      body: JSON.stringify({
        message: 'User registered successfully. Please check your email to confirm.',
        email
      })
    };
  } catch (error: any) {
    console.error('Registration error:', error);
    return {
      statusCode: 400,
      body: JSON.stringify({ error: error.message })
    };
  }
};

/**
 * Login user with email and password
 */
export const login = async (event: any) => {
  try {
    const { email, password } = JSON.parse(event.body) as AuthRequest;

    if (!email || !password) {
      return {
        statusCode: 400,
        body: JSON.stringify({ error: 'Email and password required' })
      };
    }

    // Authenticate with Cognito
    const params = {
      AuthFlow: 'USER_PASSWORD_AUTH',
      ClientId: process.env.COGNITO_CLIENT_ID!,
      AuthParameters: {
        USERNAME: email,
        PASSWORD: password
      }
    };

    const authResult = await cognito
      .initiateAuth(params)
      .promise();

    if (!authResult.AuthenticationResult) {
      return {
        statusCode: 401,
        body: JSON.stringify({ error: 'Authentication failed' })
      };
    }

    // Get user details from DynamoDB
    const userResult = await dynamodb.query({
      TableName: process.env.USERS_TABLE!,
      IndexName: 'emailIndex',
      KeyConditionExpression: 'email = :email',
      ExpressionAttributeValues: {
        ':email': email
      }
    }).promise();

    const user = userResult.Items?.[0];

    return {
      statusCode: 200,
      body: JSON.stringify({
        message: 'Login successful',
        user: {
          id: user?.id,
          email: user?.email,
          name: user?.name,
          subscriptionTier: user?.subscriptionTier,
          role: user?.role,
          level: user?.level,
          photo: user?.photo
        },
        tokens: {
          idToken: authResult.AuthenticationResult.IdToken,
          accessToken: authResult.AuthenticationResult.AccessToken,
          refreshToken: authResult.AuthenticationResult.RefreshToken
        }
      })
    };
  } catch (error: any) {
    console.error('Login error:', error);
    return {
      statusCode: 401,
      body: JSON.stringify({ error: error.message })
    };
  }
};

/**
 * Verify JWT token and return user info
 */
export const verify = async (event: any) => {
  try {
    const token = event.headers.Authorization?.replace('Bearer ', '');

    if (!token) {
      return {
        statusCode: 401,
        body: JSON.stringify({ error: 'No token provided' })
      };
    }

    // Get user from Cognito using token
    const params = {
      AccessToken: token
    };

    const userInfo = await cognito.getUser(params).promise();

    const email = userInfo.UserAttributes?.find(
      (attr) => attr.Name === 'email'
    )?.Value;

    // Get full user details from DynamoDB
    const userResult = await dynamodb.query({
      TableName: process.env.USERS_TABLE!,
      IndexName: 'emailIndex',
      KeyConditionExpression: 'email = :email',
      ExpressionAttributeValues: {
        ':email': email

    }).promise();

    const user = userResult.Items?.[0];

    return {
      statusCode: 200,

/**
 * Send announcement via Gmail
 * POST /api/announcements/send
 */
export const sendViaGmail = async (event: any) => {
  try {
    const { title, content, recipients, accessToken } = JSON.parse(event.body);

    if (!recipients || recipients.length === 0) {
      return {
        statusCode: 400,
        body: JSON.stringify({ error: 'No recipients provided' })
      };
    }

    const auth = new google.auth.OAuth2();
    auth.setCredentials({ access_token: accessToken });

    let successful = 0;
    let failed = 0;

    // Send email to each recipient
    for (const email of recipients) {
      try {
        const message = {
          to: email,
          subject: title,
          html: content
        };

        const encodedMessage = Buffer.from(
          `From: ${process.env.SENDER_EMAIL}\nTo: ${email}\nSubject: ${title}\n\n${content}`
        ).toString('base64');

        await gmail.users.messages.send(
          {
            userId: 'me',
            requestBody: {
              raw: encodedMessage
            },
            auth
          },
          (err: any, res: any) => {
            if (err) {
              failed++;
              console.error(`Failed to send to ${email}:`, err);
            } else {
              successful++;
            }
          }
        );
      } catch (err) {
        failed++;
        console.error(`Error sending to ${email}:`, err);
      }
    }

    // Save announcement to DynamoDB
    const announcement = {
      id: crypto.randomUUID(),
      title,
      content,
      recipients,
      sentVia: 'gmail',
      createdAt: new Date().toISOString()
    };

    await dynamodb
      .put({
        TableName: process.env.ANNOUNCEMENTS_TABLE!,
        Item: announcement
      })
      .promise();

    return {
      statusCode: 200,
      body: JSON.stringify({
        announcement,
        results: { successful, failed }
      })
    };
  } catch (error: any) {
    console.error('Send announcement error:', error);
    return {
      statusCode: 400,
      body: JSON.stringify({ error: error.message })
    };
  }
};

    const { email, amount, userId } = JSON.parse(event.body);

    const reference = crypto.randomBytes(16).toString('hex');

    const payload = {
      email,
      amount: amount * 100, // Convert to kobo
      reference,
      metadata: {
        userId,
        subscriptionTier: 'premium',
        subscriptionDuration: 30
      }
    };

    const response = await axios.post(
      `${PAYSTACK_BASE_URL}/transaction/initialize`,
      payload,
      {
        headers: {
          Authorization: `Bearer ${process.env.PAYSTACK_SECRET_KEY}`
        }
      }
    );

    return {
      statusCode: 200,
      body: JSON.stringify({
        reference,
        authorizationUrl: response.data.data.authorization_url
      })
    };
  } catch (error: any) {
    console.error('Paystack initialization error:', error);
    return {
      statusCode: 400,
      body: JSON.stringify({
        error: error.response?.data || error.message
      })
    };
  }
};

/**
 * Verify Paystack payment
 * GET /api/payments/paystack/verify/{reference}
 */
export const verifyPaystack = async (event: any) => {
  try {
    const { reference } = event.pathParameters;

    const response = await axios.get(
      `${PAYSTACK_BASE_URL}/transaction/verify/${reference}`,
      {
        headers: {
          Authorization: `Bearer ${process.env.PAYSTACK_SECRET_KEY}`
        }
      }
    );

    const data = response.data.data;

    if (data.status === 'success') {
      const { userId } = data.metadata;

      // Get user
      const userResult = await dynamodb
        .get({
          TableName: process.env.USERS_TABLE!,
          Key: { id: userId }
        })
        .promise();

      // Create subscription
      const subscriptionEndDate = new Date();
      subscriptionEndDate.setDate(subscriptionEndDate.getDate() + 30);

      await dynamodb
        .put({
          TableName: process.env.SUBSCRIPTIONS_TABLE!,
          Item: {
            userId,
            tier: 'premium',
            startDate: new Date().toISOString(),
            endDate: subscriptionEndDate.toISOString(),
            isActive: true,
            paymentReference: reference,
            paymentGateway: 'paystack',
            createdAt: new Date().toISOString()
          }
        })
        .promise();

      // Update user subscription tier
      await dynamodb
        .update({
          TableName: process.env.USERS_TABLE!,
          Key: { id: userId },
          UpdateExpression: 'SET subscriptionTier = :tier',
          ExpressionAttributeValues: {
            ':tier': 'premium'
          }
        })
        .promise();

      return {
        statusCode: 200,
        body: JSON.stringify({
          status: 'success',
          message: 'Payment verified and subscription activated',
          reference
        })
      };
    }

    return {
      statusCode: 400,
      body: JSON.stringify({
        status: 'failed',
        message: 'Payment verification failed'
      })
    };
  } catch (error: any) {
    console.error('Paystack verification error:', error);
    return {
      statusCode: 400,
      body: JSON.stringify({
        error: error.response?.data || error.message
      })
    };
  }
};

/**
 * Check subscription status
 * GET /api/payments/subscription/{userId}
 */
export const checkSubscription = async (event: any) => {
  try {
    const { userId } = event.pathParameters;

    const result = await dynamodb
      .query({
        TableName: process.env.SUBSCRIPTIONS_TABLE!,
        KeyConditionExpression: 'userId = :userId',
        ExpressionAttributeValues: {
          ':userId': userId
        },
        ScanIndexForward: false,
        Limit: 1
      })
      .promise();

    const subscription = result.Items?.[0];

    if (!subscription) {
      return {
        statusCode: 200,
        body: JSON.stringify({
          tier: 'free',
          isActive: false
        })
      };
    }

    const isActive =
      subscription.isActive &&
      new Date(subscription.endDate) > new Date();

    return {
      statusCode: 200,
      body: JSON.stringify({
        tier: subscription.tier,
        isActive,
        startDate: subscription.startDate,
        endDate: subscription.endDate,
        paymentGateway: subscription.paymentGateway
      })
    };
  } catch (error: any) {
    console.error('Check subscription error:', error);
    return {
      statusCode: 400,
      body: JSON.stringify({ error: error.message })
    };
  }
};

#!/bin/bash
# deploy.sh - Complete deployment script

set -e

echo "🚀 Starting Udodiri Young Social Club Deployment..."

# Colors for output
GREEN='\033[0;32m'
BLUE='\033[0;34m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# Configuration
STACK_NAME="udodiri-club-stack"
AWS_REGION=${AWS_REGION:-us-east-1}
ENVIRONMENT=${ENVIRONMENT:-prod}

echo -e "${BLUE}Configuration:${NC}"
echo "Stack Name: $STACK_NAME"
echo "Region: $AWS_REGION"
echo "Environment: $ENVIRONMENT"

# Step 1: Build Lambda functions
echo -e "${BLUE}\n📦 Building Lambda functions...${NC}"
cd backend/lambda
npm install
npm run build:lambda
cd ../../

# Step 2: Deploy with SAM
echo -e "${BLUE}\n🚀 Deploying AWS Stack with SAM...${NC}"
sam validate --template template.yaml
sam build --template template.yaml
sam deploy \
  --template-file .aws-sam/build/template.yaml \
  --stack-name $STACK_NAME \
  --parameter-overrides Environment=$ENVIRONMENT \
  --capabilities CAPABILITY_IAM CAPABILITY_NAMED_IAM \
  --region $AWS_REGION \
  --no-confirm-changeset

# Step 3: Get API endpoint
echo -e "${BLUE}\n📋 Retrieving API Endpoint...${NC}"
API_ENDPOINT=$(aws cloudformation describe-stacks \
  --stack-name $STACK_NAME \
  --query 'Stacks[0].Outputs[?OutputKey==`ApiEndpoint`].OutputValue' \
  --output text \
  --region $AWS_REGION)

echo -e "${GREEN}✅ API Endpoint: $API_ENDPOINT${NC}"

# Step 4: Build frontend
echo -e "${BLUE}\n🏗️  Building React frontend...${NC}"
npm run build

# Step 5: Deploy to Vercel (optional)
if command -v vercel &> /dev/null; then
  echo -e "${BLUE}\n🌐 Deploying to Vercel...${NC}"
  vercel --prod \
    --env REACT_APP_API_ENDPOINT=$API_ENDPOINT
  echo -e "${GREEN}✅ Frontend deployed to Vercel${NC}"
fi

echo -e "${GREEN}\n🎉 Deployment complete!${NC}"
echo -e "${YELLOW}Next steps:${NC}"
echo "1. Update .env.production with API_ENDPOINT: $API_ENDPOINT"
echo "2. Test all features"
echo "3. Monitor CloudWatch logs"
echo "4. Share deployment URL with team"
